/* ══════════════════════════════════════════
   FIRESTORE SECURITY RULES v5.2
   ── الملف الكامل: firestore.rules ──
   ✅ Audit Logs دائمة لا تُحذف
   ✅ actorRole يُتحقق منه server-side
   ✅ Soft Delete للرسائل
   ✅ Owner يقرأ privateChats و messages بدون تعديل

firestore.rules — انسخ هذا في Firebase Console:

rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {

    // ── دوال مساعدة ──
    function isOwner() {
      return request.auth != null &&
        request.auth.uid == "wnvQPme3u0NU4VVqvr4DGqvv4KE3";
    }
    function isAdmin() {
      return request.auth != null && (
        isOwner() ||
        get(/databases/$(database)/documents/users/$(request.auth.uid)).data.role == "admin"
      );
    }
    function isSignedIn() { return request.auth != null; }

    // ── المستخدمون ──
    match /users/{uid} {
      allow read: if isSignedIn();
      allow write: if isOwner() || request.auth.uid == uid;
    }

    // ── الشات العام ──
    match /messages/{msgId} {
      allow read: if isSignedIn();
      allow create: if isSignedIn();
      allow update, delete: if isAdmin() || request.auth.uid == resource.data.uid;
    }

    // ── المحادثات الخاصة ──
    match /privateChats/{chatId} {
      // ✅ المالك يقرأ فقط — لا يكتب
      allow read: if isOwner() ||
        (isSignedIn() && request.auth.uid in resource.data.participants);
      allow write: if isSignedIn() &&
        request.auth.uid in request.resource.data.participants;

      match /messages/{msgId} {
        // ✅ المالك يقرأ فقط
        allow read: if isOwner() ||
          (isSignedIn() && request.auth.uid in
            get(/databases/$(database)/documents/privateChats/$(chatId)).data.participants);
        allow create: if isSignedIn() &&
          request.auth.uid in
            get(/databases/$(database)/documents/privateChats/$(chatId)).data.participants;
        allow update, delete: if isAdmin() || request.auth.uid == resource.data.uid;
      }
    }

    // ── الغرف ──
    match /rooms/{roomId} {
      allow read: if isSignedIn();
      allow write: if isAdmin();
      match /messages/{msgId} {
        allow read: if isSignedIn();
        allow create: if isSignedIn();
        allow update, delete: if isAdmin() || request.auth.uid == resource.data.uid;
      }
    }

    // ── Audit Logs ──
    match /auditLogs/{logId} {
      allow read: if isAdmin();
      allow create: if isAdmin();
      allow update, delete: if false;
    }

    // ── الإشعارات ──
    match /notifications/{uid}/{notifId} {
      allow read, write: if isOwner() || request.auth.uid == uid;
    }

    // ══════════════════════════════════════
    // LOST & FOUND — راحت فين؟
    // ══════════════════════════════════════
    match /lostFound/{postId} {

      allow read: if isSignedIn() && (
        resource.data.status in ["published", "found"] ||
        resource.data.createdBy == request.auth.uid ||
        isAdmin()
      );

      allow create: if isSignedIn() && (
        // مستخدم عادي: طلب جديد دايمًا pending
        (request.resource.data.createdBy == request.auth.uid &&
         request.resource.data.status == "pending" &&
         request.resource.data.type in ["lost", "found"] &&
         request.resource.data.title is string &&
         request.resource.data.title.size() > 0)
        ||
        // الأدمن: نشر مباشر — status == published من الإنشاء مباشرة
        (isAdmin() &&
         request.resource.data.createdBy == request.auth.uid &&
         request.resource.data.status == "published" &&
         request.resource.data.approvedBy == request.auth.uid &&
         request.resource.data.approvedAt == request.time &&
         request.resource.data.type in ["lost", "found"] &&
         request.resource.data.title is string &&
         request.resource.data.title.size() > 0)
      );

      allow update: if isSignedIn() && (
        // ── الأدمن: كل انتقال مقيّد بحقوله بالظبط ──

        // 1) موافقة: pending → published
        (isAdmin() &&
         resource.data.status == "pending" &&
         request.resource.data.status == "published" &&
         request.resource.data.approvedBy == request.auth.uid &&
         request.resource.data.diff(resource.data).affectedKeys().hasOnly(["status","approvedBy","approvedAt"])) ||

        // 2) رفض: pending → rejected
        (isAdmin() &&
         resource.data.status == "pending" &&
         request.resource.data.status == "rejected" &&
         request.resource.data.rejectedBy == request.auth.uid &&
         request.resource.data.diff(resource.data).affectedKeys().hasOnly(["status","rejectedBy","rejectedAt","rejectionReason"])) ||

        // 3) طلب استكمال بيانات: adminNote بس، الحالة تفضل pending
        (isAdmin() &&
         resource.data.status == "pending" &&
         request.resource.data.status == "pending" &&
         request.resource.data.diff(resource.data).affectedKeys().hasOnly(["adminNote"])) ||

        // 4) تصحيح: found → published (تراجع)
        (isAdmin() &&
         resource.data.status == "found" &&
         request.resource.data.status == "published" &&
         request.resource.data.diff(resource.data).affectedKeys().hasOnly(["status","resolvedBy","resolvedAt"])) ||

        // 5) الأدمن يقفل منشور بالنيابة: published → found
        (isAdmin() &&
         resource.data.status == "published" &&
         request.resource.data.status == "found" &&
         request.resource.data.resolvedBy == request.auth.uid &&
         request.resource.data.diff(resource.data).affectedKeys().hasOnly(["status","resolvedBy","resolvedAt"])) ||

        // ── صاحب المنشور (بدون صلاحية أدمن) ──

        // 6) يعدّل بياناته وهو لسه pending أو rejected (إعادة إرسال) — حقول محددة بالظبط
        (request.auth.uid == resource.data.createdBy &&
         resource.data.status in ["pending", "rejected"] &&
         request.resource.data.status == "pending" &&
         request.resource.data.createdBy == resource.data.createdBy &&
         request.resource.data.diff(resource.data).affectedKeys().hasOnly(
           ["type","title","description","location","images",
            "status","rejectedBy","rejectedAt","rejectionReason","adminNote"]
         )) ||

        // 7) يقفل منشوره المنشور (لقيتها)
        (request.auth.uid == resource.data.createdBy &&
         resource.data.status == "published" &&
         request.resource.data.status == "found" &&
         request.resource.data.resolvedBy == request.auth.uid)
      );

      allow delete: if isAdmin() ||
        (request.auth.uid == resource.data.createdBy && resource.data.status != "found");

      match /comments/{commentId} {
        allow read: if isSignedIn();

        allow create: if isSignedIn() &&
          request.resource.data.createdBy == request.auth.uid &&
          get(/databases/$(database)/documents/lostFound/$(postId)).data.status == "published";

        allow update: if false;

        allow delete: if isAdmin() ||
          request.auth.uid == resource.data.createdBy;
      }
    }

    // ── باقي المجموعات ──
    match /{document=**} {
      allow read: if isAdmin();
      allow write: if isAdmin();
    }
  }
}
